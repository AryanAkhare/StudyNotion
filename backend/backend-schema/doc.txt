Table users {
  id varchar [primary key, note: "MongoDB ObjectId"]
  firstName varchar
  lastName varchar
  email varchar [note: "unique"]
  password varchar
  accountType varchar [note: "Admin | Student | Instructor"]
  additionalDetails varchar [note: "ref profiles.id"]
  image varchar
  courses varchar [note: "array of course ids"]
  courseProgess varchar [note: "array of course_progress ids"]
  token varchar [note: "password-reset token"]
  resetPasswordExpires datetime
}

Table profiles {
  id varchar [primary key, note: "MongoDB ObjectId"]
  gender varchar [note: "Male | Female | Other"]
  dateOfBirth datetime
  about text
  contactNumber varchar
}

Table otp {
  id varchar [primary key, note: "MongoDB ObjectId"]
  email varchar
  otp varchar
  createdAt datetime [note: "TTL: 5 minutes, pre-save sends email"]
}

Table courses {
  id varchar [primary key, note: "MongoDB ObjectId"]
  courseName varchar
  courseDescription text
  instructor varchar [note: "ref users.id"]
  whatYouWillLearn text
  price int
  category varchar [note: "ref categories.id"]
  courseContent varchar [note: "array of section ids"]
  ratingAndReviews varchar [note: "array of rating_and_reviews ids"]
  studentsEnrolled varchar [note: "array of user ids"]
}

Table sections {
  id varchar [primary key, note: "MongoDB ObjectId"]
  sectionName varchar
  courseId varchar [note: "ref courses.id"]
}

Table sub_sections {
  id varchar [primary key, note: "MongoDB ObjectId"]
  title varchar
  timeDuration varchar
  description text
  videoUrl varchar
  additionalUrl varchar
  section_id varchar [note: "ref sections.id"]
}

Table course_progress {
  id varchar [primary key, note: "MongoDB ObjectId"]
  courseId varchar [note: "ref courses.id"]
  completedVideos varchar [note: "array of sub_section ids"]
}

Table rating_and_reviews {
  id varchar [primary key, note: "MongoDB ObjectId"]
  user varchar [note: "ref users.id"]
  rating int
  review text
}

Table categories {
  id varchar [primary key, note: "MongoDB ObjectId"]
  name varchar
  description text
  course varchar [note: "ref courses.id"]
}

Table invoices {
  id varchar [primary key, note: "MongoDB ObjectId"]
  userId varchar
  courseId varchar
  courseName varchar
  price int
  billingAddress text
  pincode varchar
}

Ref: users.additionalDetails > profiles.id

Ref: courses.instructor > users.id
Ref: courses.category > categories.id

Ref: sections.courseId > courses.id
Ref: sub_sections.section_id > sections.id

Ref: course_progress.courseId > courses.id
Ref: rating_and_reviews.user > users.id

Ref: categories.course > courses.id

Ref: invoices.userId > users.id
Ref: invoices.courseId > courses.id
